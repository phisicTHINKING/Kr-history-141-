          <p style="font-size:0.88rem; color:var(--text-sub); margin-top:3px;">미국·소련 등 강대국이 "한국은 아직 혼자 나라 운영하기 힘드니 최대 5년간 우리가 대신 관리해 줄게"라고 한 계획입니다.</p>
        </div>
        <div style="background:#f8fafc; padding:12px 14px; border-radius:10px; border:1px solid var(--border);">

<script>
  function safeGetStorage(key, fallback) {
    try {
      const val = window.localStorage.getItem(key);
      return val ? JSON.parse(val) : fallback;
    } catch (e) {
      return fallback;
    }
  }


  function safeSetStorage(key, val) {
    try {
      window.localStorage.setItem(key, JSON.stringify(val));
    } catch (e) {}
  }


  let notesData = [];
  let memorizedIds = safeGetStorage('kr_history_memorized', []);
  let currentEra = 'ALL';
  let currentOrderProblem = null;
  let currentOrderState = [];
  let quizScore = 0;


  function init() {
    const savedCustom = safeGetStorage('kr_history_custom_notes', []);
    notesData = [...DEFAULT_NOTES, ...savedCustom];
    renderNotes();
    updateProgress();
    initOrderGame();
    renderQuiz();
  }


  function toggleBlindMode() {
    document.body.classList.toggle('blind-mode');
    const btn = document.getElementById('blindToggleBtn');
    const isBlind = document.body.classList.contains('blind-mode');
    btn.classList.toggle('btn-blind-active', isBlind);
    btn.textContent = isBlind ? '👀 정답 모두 보기' : '🙈 암기 모드 (빈칸 가리기)';
    document.querySelectorAll('.cloze').forEach(el => el.classList.remove('revealed'));
  }


  function parseCloze(text) {
    return text.replace(/\[([^\]]+)\]/g, `<span class="cloze" onclick="this.classList.toggle('revealed')">$1</span>`);
  }


  function filterNotes(era, btn) {
    currentEra = era;
    document.querySelectorAll('.filter-pill').forEach(p => p.classList.remove('active'));
    btn.classList.add('active');
    renderNotes();
  }


  document.getElementById('searchInput').addEventListener('input', () => {
    document.getElementById('r-tab-3').checked = true;
    renderNotes();
  });


  function renderNotes() {
    const container = document.getElementById('notesContainer');
    const keyword = document.getElementById('searchInput').value.trim().toLowerCase();


    const filtered = notesData.filter(item => {
      const matchEra = currentEra === 'ALL' || item.era === currentEra;
      const fullText = (item.title + ' ' + item.context + ' ' + item.bullets.join(' ') + ' ' + (item.tip || '')).toLowerCase();
      const matchKeyword = !keyword || fullText.includes(keyword);
      return matchEra && matchKeyword;
    });


    if (filtered.length === 0) {
      container.innerHTML = '<div class="card" style="text-align:center; color:var(--text-sub);">검색 조건에 맞는 노트가 없습니다.</div>';
      return;
    }


    const eraNames = {
      '해방공간': '해방과 분단',
      '정부수립': '정부수립·개혁',
      '625전쟁': '6·25 전쟁',
      '1_2공화국': '1·2공화국',
      '3_4공화국': '박정희·유신',
      '5_6공화국': '5·6공화국'
    };


    container.innerHTML = filtered.map(note => {
      const isDone = memorizedIds.includes(note.id);
      return `
        <div class="card" style="${isDone ? 'border-color: var(--success); background: #fcfffd;' : ''}">
          <div class="card-title">
            <div>
              <span class="era-tag" style="margin-right:6px;">${eraNames[note.era] || note.era}</span>
              <span>${note.title}</span>
            </div>
            <button type="button" class="btn" style="${isDone ? 'background:var(--success); color:#fff; border-color:var(--success);' : ''}" onclick="toggleMemorized('${note.id}')">
              ${isDone ? '✅ 암기 완료' : '⬜ 암기 체크'}
            </button>
          </div>
          <div class="note-context-box">
            <strong>💡 한 줄 맥락:</strong> ${note.context}
          </div>
          <ul class="note-list">
            ${note.bullets.map(b => `<li>${parseCloze(b)}</li>`).join('')}
          </ul>
          ${note.tip ? `<div class="exam-tip">📌 ${note.tip}</div>` : ''}
        </div>
      `;
    }).join('');
  }


  function toggleMemorized(id) {
    if (memorizedIds.includes(id)) {
      memorizedIds = memorizedIds.filter(i => i !== id);
    } else {
      memorizedIds.push(id);
    }
    safeSetStorage('kr_history_memorized', memorizedIds);
    renderNotes();
    updateProgress();
  }


  function updateProgress() {
    const total = notesData.length;
    const done = memorizedIds.filter(id => notesData.some(n => n.id === id)).length;
    document.getElementById('progressText').textContent = `${done} / ${total} 단원`;
    const pct = total ? Math.round((done / total) * 100) : 0;
    document.getElementById('progressBar').style.width = `${pct}%`;
  }


  function initOrderGame() {
    const randIdx = Math.floor(Math.random() * ORDER_PROBLEMS.length);
    currentOrderProblem = ORDER_PROBLEMS[randIdx];
    currentOrderState = [...currentOrderProblem.items].sort(() => Math.random() - 0.5);
    document.getElementById('orderFeedback').textContent = '';
    renderOrderList();
  }


  function renderOrderList() {
    const listEl = document.getElementById('orderList');
    listEl.innerHTML = `<div style="font-weight:800; margin-bottom:10px; color:var(--primary);">[문제] ${currentOrderProblem.title}</div>` +
      currentOrderState.map((item, idx) => `
        <div class="order-item">
          <span>${idx + 1}. ${item.text}</span>
          <div style="display:flex; gap:4px; flex-shrink:0;">
            <button type="button" class="btn" onclick="moveOrder(${idx}, -1)" ${idx === 0 ? 'disabled' : ''}>▲</button>
            <button type="button" class="btn" onclick="moveOrder(${idx}, 1)" ${idx === currentOrderState.length - 1 ? 'disabled' : ''}>▼</button>
          </div>
        </div>
      `).join('');
  }


  function moveOrder(idx, dir) {
    const target = idx + dir;
    if (target < 0 || target >= currentOrderState.length) return;
    const temp = currentOrderState[idx];
    currentOrderState[idx] = currentOrderState[target];
    currentOrderState[target] = temp;
    renderOrderList();
  }


  function checkOrderGame() {
    const isCorrect = currentOrderState.every((item, idx) => item.step === idx + 1);
    const fb = document.getElementById('orderFeedback');
    if (isCorrect) {
      fb.style.color = 'var(--success)';
      fb.textContent = '🎉 완벽합니다! 시대 흐름이 정확하게 정렬되었습니다.';
    } else {
      fb.style.color = 'var(--danger)';
      fb.textContent = '❌ 순서가 틀린 곳이 있습니다. 2번 탭의 연상 공식을 떠올려 보세요!';
    }
  }


  function renderQuiz() {
    quizScore = 0;
    updateQuizBadge();
    const container = document.getElementById('quizContainer');
    container.innerHTML = QUIZ_QUESTIONS.map((qObj, qIdx) => `
      <div style="margin-bottom:22px; padding-bottom:18px; border-bottom:1px solid var(--border);">
        <div style="font-weight:700; margin-bottom:10px; font-size:0.98rem;">${qObj.q}</div>
        <div id="q-opts-${qIdx}">
          ${qObj.options.map((opt, oIdx) => `
            <button type="button" class="quiz-option" onclick="answerQuiz(${qIdx}, ${oIdx}, this)">${oIdx + 1}) ${opt}</button>
          `).join('')}
        </div>
        <div id="q-exp-${qIdx}" style="display:none; margin-top:8px; padding:10px 12px; background:#f1f5f9; border-radius:8px; font-size:0.88rem; color:var(--text-sub);"></div>
      </div>
    `).join('');
  }


  function answerQuiz(qIdx, oIdx, btn) {
    const optsWrap = document.getElementById(`q-opts-${qIdx}`);
    if (optsWrap.dataset.answered) return;
    optsWrap.dataset.answered = 'true';


    const qObj = QUIZ_QUESTIONS[qIdx];
    const buttons = optsWrap.querySelectorAll('.quiz-option');


    if (oIdx === qObj.answer) {
      btn.classList.add('correct');
      quizScore++;
    } else {
      btn.classList.add('wrong');
      buttons[qObj.answer].classList.add('correct');
    }


    const expEl = document.getElementById(`q-exp-${qIdx}`);
    expEl.style.display = 'block';
    expEl.innerHTML = `<strong>해설:</strong> ${qObj.exp}`;
    updateQuizBadge();
  }


  function updateQuizBadge() {
    document.getElementById('quizScoreBadge').textContent = `점수: ${quizScore} / ${QUIZ_QUESTIONS.length}`;
  }


  function addNewNote() {
    const era = document.getElementById('newEra').value;
    const title = document.getElementById('newTitle').value.trim();
    const context = document.getElementById('newContext').value.trim();
    const bulletsRaw = document.getElementById('newBullets').value.trim();
    const tip = document.getElementById('newTip').value.trim();


    if (!title || !bulletsRaw) {
      alert('제목과 핵심 암기 내용을 입력해 주세요!');
      return;
    }


    const newItem = {
      id: 'custom-' + Date.now(),
      era,
      title,
      context: context || '추가된 학습 자료 핵심 정리',
      bullets: bulletsRaw.split('\n').map(line => line.trim()).filter(Boolean),
      tip
    };


    const savedCustom = safeGetStorage('kr_history_custom_notes', []);
    savedCustom.push(newItem);
    safeSetStorage('kr_history_custom_notes', savedCustom);


    notesData.push(newItem);
    renderNotes();
    updateProgress();


    document.getElementById('newTitle').value = '';
    document.getElementById('newContext').value = '';
    document.getElementById('newBullets').value = '';
    document.getElementById('newTip').value = '';


    alert('✅ 정리노트에 추가되었습니다! 깃허브에도 반영하려면 아래 [업데이트된 index.html 다운로드] 버튼을 눌러주세요.');
  }


  function exportUpdatedHTML() {
    const docClone = document.documentElement.cloneNode(true);
    const dataScript = docClone.querySelector('#app-data-script');
    dataScript.textContent = `
const DEFAULT_NOTES = ${JSON.stringify(notesData, null, 2)};
const ORDER_PROBLEMS = ${JSON.stringify(ORDER_PROBLEMS, null, 2)};
const QUIZ_QUESTIONS = ${JSON.stringify(QUIZ_QUESTIONS, null, 2)};
    `;
    docClone.querySelector('#orderList').innerHTML = '';
    docClone.querySelector('#quizContainer').innerHTML = '';


    const htmlContent = '<!DOCTYPE html>\n' + docClone.outerHTML;
    const blob = new Blob([htmlContent], { type: 'text/html;charset=utf-8' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = 'index.html';
    a.click();
    URL.revokeObjectURL(url);
  }
